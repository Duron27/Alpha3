cmake_minimum_required(VERSION 3.8)
project(angle CXX C)

set(CMAKE_BUILD_TYPE "Release")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)

# Fetch ANGLE source
FetchContent_Declare(
    angle_src
    GIT_REPOSITORY https://chromium.googlesource.com/angle/angle
    GIT_TAG        chromium/6536
    GIT_SHALLOW    TRUE
    GIT_SUBMODULES ""
    GIT_PROGRESS   TRUE
)

# Fetch dependencies
FetchContent_Declare(
    rapidjson
    GIT_REPOSITORY https://github.com/TenCent/rapidjson.git
    GIT_TAG        master
    GIT_PROGRESS   TRUE
)

FetchContent_Declare(
    zlib_repo
    GIT_REPOSITORY https://chromium.googlesource.com/chromium/src/third_party/zlib.git
    GIT_TAG        main
    GIT_PROGRESS   TRUE
)

FetchContent_Declare(
    spirv_headers
    GIT_REPOSITORY https://github.com/KhronosGroup/SPIRV-Headers.git
    GIT_TAG        main
    GIT_PROGRESS   TRUE
)

FetchContent_Declare(
    spirv_tools
    GIT_REPOSITORY https://github.com/KhronosGroup/SPIRV-Tools.git
    GIT_TAG        main
    GIT_PROGRESS   TRUE
)

FetchContent_Declare(
    vulkan_headers
    GIT_REPOSITORY https://github.com/KhronosGroup/Vulkan-Headers.git
    GIT_TAG        main
    GIT_PROGRESS   TRUE
)

FetchContent_Declare(
    vulkan_memory_allocator
    GIT_REPOSITORY https://github.com/GPUOpen-LibrariesAndSDKs/VulkanMemoryAllocator.git
    GIT_TAG        master
    GIT_PROGRESS   TRUE
)

set(FETCHCONTENT_QUIET OFF)

# Configure dependencies
set(ZLIB_CONF_WRITTEN OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)

FetchContent_MakeAvailable(zlib_repo)

set(RAPIDJSON_BUILD_DOC OFF CACHE BOOL "" FORCE)
set(RAPIDJSON_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(RAPIDJSON_BUILD_TESTS OFF CACHE BOOL "" FORCE)

FetchContent_MakeAvailable(rapidjson)

# Make ANGLE source available
FetchContent_MakeAvailable(angle_src)

# Set ANGLE source directory after FetchContent
set(ANGLE_SRC_DIR "${angle_src_SOURCE_DIR}")
set(ANGLE_PATCH_SRC_DIR "${CMAKE_CURRENT_LIST_DIR}")

# Add subdirectories if they exist
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/cmake_tools/compression_utils_portable")
    add_subdirectory(cmake_tools/compression_utils_portable)
endif()
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/cmake_tools/shim_synchapi")
    add_subdirectory(cmake_tools/shim_synchapi)
endif()

set(SPIRV_SKIP_TESTS ON CACHE BOOL "" FORCE)
set(SPIRV_SKIP_EXECUTABLES ON CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)

FetchContent_MakeAvailable(spirv_headers)
FetchContent_MakeAvailable(spirv_tools)
FetchContent_MakeAvailable(vulkan_headers)
FetchContent_MakeAvailable(vulkan_memory_allocator)

# Find Android system libraries
find_library(ANDROID_LOG_LIB log)
find_library(ANDROID_LIB android)
find_library(ANDROID_NATIVE_WINDOW_LIB nativewindow)

set(CMAKE_MODULE_LINKER_FLAGS "${CMAKE_MODULE_LINKER_FLAGS}")
set(CMAKE_STATIC_LIBRARY_PREFIX "")

# Android-specific definitions
add_definitions(
    -D_CRT_SECURE_NO_DEPRECATE
    -D_SCL_SECURE_NO_WARNINGS
    -D_HAS_EXCEPTIONS=0
    -DNOMINMAX
    -DANGLE_STANDALONE_BUILD
    -DANGLE_ENABLE_DEBUG_ANNOTATIONS
    -DANGLE_PLATFORM_ANDROID
    -DANGLE_ENABLE_ANDROID_LOGGING
    -DANGLE_ENABLE_VULKAN
    -DVK_USE_PLATFORM_ANDROID_KHR
)

# Create build include dir for autogenerated headers
set(AUTO_GENERATED_INCLUDE_DIR ${CMAKE_CURRENT_BINARY_DIR}/include)
file(MAKE_DIRECTORY ${AUTO_GENERATED_INCLUDE_DIR})

# Generate commit header if the script exists
if(EXISTS "${ANGLE_SRC_DIR}/src/commit_id.py")
    execute_process(
        COMMAND python3 ${ANGLE_SRC_DIR}/src/commit_id.py gen ${CMAKE_CURRENT_BINARY_DIR}/include/angle_commit.h
        OUTPUT_VARIABLE IS_GIT_REPO
        OUTPUT_STRIP_TRAILING_WHITESPACE
    )
endif()

include_directories(${ANGLE_SRC_DIR}/src ${CMAKE_CURRENT_BINARY_DIR}/include)

    add_library(angle_common STATIC ${ANGLE_COMMON_SOURCES})

    add_library(libGLESv2 SHARED ${LIBGLESV2_SOURCES})
    target_compile_definitions(libGLESv2 PUBLIC "-DEGLAPI=__attribute__\(\(visibility\(\"default\"\)\)\)")
    target_compile_definitions(libGLESv2 PUBLIC "-DGL_APICALL=__attribute__\(\(visibility\(\"default\"\)\)\)")
    target_link_libraries(libGLESv2 PRIVATE angle_common libANGLE)
    target_compile_definitions(libGLESv2
        PRIVATE -DLIBGLESV2_IMPLEMENTATION
        PUBLIC
            -DGL_GLEXT_PROTOTYPES
            -DGL_GLES_PROTOTYPES
    )
endif()
